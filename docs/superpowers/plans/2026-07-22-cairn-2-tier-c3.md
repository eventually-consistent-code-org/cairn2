**Interfaces:**

- Produces:
  ```ts
  export interface PlanFinding {
    type: "contract-drift" | "unanchored-threshold";
    plan: string;          // path relative to projectDir
    line: number;          // 1-indexed
    detail: string;
    counterpart?: { plan: string; line: number };
  }
  export function planCheck(projectDir: string, phase?: number):
    { findings: PlanFinding[]; scanned: number };
  ```

**Detector rules (implement exactly):**
- Scan every `PLAN.md` under `.cairn/plans/phases/*/` (filter to the single matching `NN-` dir when `phase` given — same zero-padded prefix match `status.ts` uses).
- **Contracts:** a contract line is `- Produces: <text>` or `- Consumes: <text>` (case-sensitive keys, optional leading whitespace); its text continues through immediately following lines that are more-indented or inside a code fence opened on/after the contract line, ending at the next `- ` bullet at the same indent or blank-line + outdent. For each contract, extract symbols: every `` `identifier` `` and every `identifier(` token (identifier = `[A-Za-z_$][\w$]*`, dedup per contract). **Drift:** symbol S appears in a `Produces` contract of plan A and a `Consumes` contract of plan B (A ≠ B), and the normalized contract texts differ (collapse whitespace runs to one space, trim), and NO shared fixture exists — a shared fixture is any path-like token (`/\S+\.(?:md|json|ts|js|mjs|csv|txt)/`) present in BOTH plans' full text. Finding on the CONSUMER line, counterpart = producer.
- **Thresholds:** a threshold is a match of `/(?:<=|>=|<|>|≤|≥|under|over|at least|at most|within)\s*\d+(?:\.\d+)?\s*(?:ms|s|%|rps|qps|kb|mb|gb|tok|tokens|rows|items)\b/i` on a plan line. It is **anchored** (silent) when the same line or the line directly above/below contains a path-like token (regex above) or one of the anchor words `/\b(?:benchmark|fixture|measured|per spec|spec §|source:)\b/i`. Unanchored → finding with the matched text in `detail`.
- Sort findings by (plan, line); `scanned` = number of PLAN.md files read. Missing plans root → `{ findings: [], scanned: 0 }`.

```ts
import { existsSync, mkdirSync, readFileSync, readdirSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { CairnError } from "../errors.js";
import { parseFrontmatter, serializeFrontmatter } from "../planning/frontmatter.js";

export type AuditSeverity = "critical" | "important" | "minor";
export interface AuditFinding { severity: AuditSeverity; title: string; detail?: string; issue?: string; }

const auditDir = (p: string) => join(p, ".cairn", "audit");
const today = () => new Date().toISOString().slice(0, 10);
const SEVERITIES: AuditSeverity[] = ["critical", "important", "minor"];

export function writeAuditRecord(projectDir: string, scope: string,
  verdict: "pass" | "findings", findings: AuditFinding[]): { path: string; findings: number } {
  if (!/^[a-z0-9][a-z0-9-]*$/.test(scope)) {
    throw new CairnError("UNSUPPORTED", `audit scope '${scope}' is empty or not kebab-case`,
      "use a short kebab-case scope like uat-phase-1");
  }
  if (verdict === "pass" && findings.length > 0) {
    throw new CairnError("PRECONDITION_FAILED",
      "verdict 'pass' with findings attached — pick one",
      "verdict must be 'findings' when any finding exists");
  }
  for (const f of findings) {
    if (!SEVERITIES.includes(f.severity) || f.title.trim().length === 0) {
      throw new CairnError("UNSUPPORTED", "finding needs a severity (critical|important|minor) and a title", "");
    }
  }
  const body = [`# Audit: ${scope}`, ""];
  for (const f of findings) {
    body.push(`## finding — ${f.severity}`, f.title);
    if (f.issue) body.push(`issue: ${f.issue}`);
    if (f.detail) body.push("", f.detail.trimEnd());
    body.push("");
  }
  mkdirSync(auditDir(projectDir), { recursive: true });
  const path = join(auditDir(projectDir), `${scope}-${today()}.md`);
  writeFileSync(path, serializeFrontmatter(
    { scope, verdict, created: today() }, `${body.join("\n").trimEnd()}\n`));
  return { path, findings: findings.length };
}

export function listAuditRecords(projectDir: string):
  Array<{ scope: string; date: string; verdict: string; path: string }> {
  const dir = auditDir(projectDir);
  if (!existsSync(dir)) return [];
  const out = [];
  for (const entry of readdirSync(dir).sort()) {
    if (!entry.endsWith(".md")) continue;
    try {
      const { data } = parseFrontmatter(readFileSync(join(dir, entry), "utf8"));
      out.push({ scope: String(data.scope ?? ""), date: String(data.created ?? ""),
        verdict: String(data.verdict ?? ""), path: join(dir, entry) });
    } catch { /* malformed record: skip, list must not brick (cards precedent) */ }
  }
  return out;
}
```
